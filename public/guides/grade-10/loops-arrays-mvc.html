Diseñar, desarrollar, documentar y presentar una aplicación web interactiva funcional (juego, simulador o herramienta de gestión de datos) utilizando JavaScript, HTML y CSS que resuelva un problema real en la escuela o comunidad, aplicando conceptos de Tecnología, Matemáticas, Español y Ciencias Sociales y promoviendo los ODS.

js/controller.js

function rollOnce() { const value = DiceModel.rollDice(); DiceView.showDice(value); DiceView.showHistory(DiceModel.rolls); } document.getElementById("rollBtn") .addEventListener("click", rollOnce); document.getElementById("roll10Btn") .addEventListener("click", () => { for (let i = 0; i < 10; i++) { rollOnce(); } }); document.getElementById("resetBtn") .addEventListener("click", () => { DiceModel.reset(); DiceView.showDice("🎲"); DiceView.showHistory([]); });
