<script>

//========= MODELO ==========
class Modelo{
constructor(){
this.puntos=0;
}
}

//========= VISTA ==========
class Vista{
actualizar(p){
document.getElementById("puntos").textContent=p;
}
}

//========= CONTROLADOR ==========
class Controlador{

constructor(){
this.modelo=new Modelo();
this.vista=new Vista();

this.jugador=document.getElementById("jugador");
this.area=document.getElementById("juego");

this.x=165;

document.addEventListener("keydown",e=>{

if(e.key=="ArrowLeft" && this.x>0)
this.x-=20;

if(e.key=="ArrowRight" && this.x<330)
this.x+=20;

this.jugador.style.left=this.x+"px";

});

setInterval(()=>this.crearObjeto(),1000);

}

crearObjeto(){

let obj=document.createElement("div");
obj.className="objeto";

let reciclable=Math.random()>0.3;

obj.innerHTML=reciclable?"♻️":"🔋";

let x=Math.random()*360;

obj.style.left=x+"px";
obj.style.top="0px";

this.area.appendChild(obj);

let y=0;

let caer=setInterval(()=>{

y+=5;
obj.style.top=y+"px";

if(y>460){

let jugadorX=this.x;

if(x>jugadorX-20 && x<jugadorX+70){

if(reciclable){
this.modelo.puntos+=10;
}else{
this.modelo.puntos-=5;
}

this.vista.actualizar(this.modelo.puntos);

}

obj.remove();
clearInterval(caer);

}

},40);

}

}

new Controlador();

</script>
